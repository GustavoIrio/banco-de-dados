-- BiblioTech — estrutura criada na lista 4 (4 tabelas)
-- Para o MySQL de onecompiler.com/mysql: cole tudo no editor e clique Run.
-- Cada Run começa com o banco vazio: deixe estes comandos no topo do

CREATE TABLE Alunos (
    AlunoID   INT AUTO_INCREMENT PRIMARY KEY,
    Nome      VARCHAR(100) NOT NULL,
    Matricula CHAR(6)      NOT NULL UNIQUE,
    Turma     VARCHAR(10)  NOT NULL,
    Email     VARCHAR(100)
);
CREATE TABLE Autores (
    AutorID       INT AUTO_INCREMENT PRIMARY KEY,
    Nome          VARCHAR(100) NOT NULL,
    Nacionalidade VARCHAR(50)
);
CREATE TABLE Livros (
    LivroID       INT AUTO_INCREMENT PRIMARY KEY,
    AutorID       INT          NOT NULL,
    Titulo        VARCHAR(150) NOT NULL,
    ISBN          CHAR(13)     NOT NULL UNIQUE,
    AnoPublicacao INT,
    Exemplares    INT          NOT NULL DEFAULT 1,
    CONSTRAINT fk_livros_autor FOREIGN KEY (AutorID) REFERENCES Autores(AutorID)
);

CREATE TABLE Emprestimos (
    EmprestimoID   INT AUTO_INCREMENT PRIMARY KEY,
    AlunoID        INT  NOT NULL,
    LivroID        INT  NOT NULL,
    DataEmprestimo DATE NOT NULL,
    DataDevolucao  DATE NULL,
    CONSTRAINT fk_emp_aluno FOREIGN KEY (AlunoID) REFERENCES Alunos(AlunoID),
    CONSTRAINT fk_emp_livro FOREIGN KEY (LivroID) REFERENCES Livros(LivroID)
);

Parte B — Consultar

Uma consulta por item. Anote quantas linhas vieram.

  1. Título e ano dos livros publicados antes de 1950, do mais antigo para o mais novo. básico
  2. Nome dos alunos que não têm e-mail cadastrado. básico
  3. Título e exemplares dos livros que têm de 2 a 4 exemplares, em ordem alfabética. básico
  4. Nome e turma dos alunos das turmas do primeiro ano (turmas que começam com DS-1). básico
  5. Títulos que contêm "Banco de Dados". básico
  6. Os empréstimos ainda não devolvidos (todas as colunas). básico
  7. As nacionalidades dos autores, sem repetição, em ordem alfabética. intermediário
  8. O livro mais antigo do acervo (só ele). intermediário
  9. Empréstimos feitos em abril ou maio de 2024 pelos alunos de ID 2 ou 4. intermediário
